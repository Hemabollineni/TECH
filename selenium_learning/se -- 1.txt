# TC02: Search iPhone and verify top 5 results
def test_iphone_search_results(driver):

    driver.get(URL)

    wait = WebDriverWait(driver, 15)

    search_box = wait.until(
        EC.visibility_of_element_located(
            (By.ID, "twotabsearchtextbox")
        )
    )

    search_box.send_keys("iPhone")

    driver.find_element(
        By.ID, "nav-search-submit-button"
    ).click()

    results = wait.until(
        EC.presence_of_all_elements_located(
            (By.CSS_SELECTOR,
             "[data-component-type='s-search-result']")
        )
    )

    # Get product titles
    product_titles = []

    for result in results:

        title_element = result.find_elements(
            By.CSS_SELECTOR, "h2 span"
        )

        if title_element:
            product_titles.append(title_element[0].text)

        if len(product_titles) == 5:
            break

    print("Top 5 search results:")

    for title in product_titles:
        print(title)

    assert len(product_titles) == 5

    # Verify all five results are iPhone-related
    for title in product_titles:
        assert "iphone" in title.lower(), (
            f"Unrelated product found: {title}"
        )


# TC03: Verify sponsored iPhone-related products
def test_sponsored_products(driver):

    driver.get(URL)

    wait = WebDriverWait(driver, 15)

    search_box = wait.until(
        EC.visibility_of_element_located(
            (By.ID, "twotabsearchtextbox")
        )
    )

    search_box.send_keys("iPhone")

    driver.find_element(
        By.ID, "nav-search-submit-button"
    ).click()

    wait.until(
        EC.presence_of_element_located(
            (By.CSS_SELECTOR,
             "[data-component-type='s-search-result']")
        )
    )

    sponsored_products = []

    results = driver.find_elements(
        By.CSS_SELECTOR,
        "[data-component-type='s-search-result']"
    )

    for result in results:

        sponsored_labels = result.find_elements(
            By.XPATH,
            ".//*[contains(translate(normalize-space(.), "
            "'SPONSORED', 'sponsored'), 'sponsored')]"
        )

        if sponsored_labels:

            titles = result.find_elements(
                By.CSS_SELECTOR, "h2 span"
            )

            if titles and titles[0].text.strip():
                sponsored_products.append(titles[0].text)

    print("Sponsored products:")

    for title in sponsored_products:
        print(title)

    if sponsored_products:

        for title in sponsored_products:
            assert "iphone" in title.lower(), (
                f"Unrelated sponsored product: {title}"
            )

        print("Sponsored products are related to iPhone")

    else:
        pytest.skip("No sponsored products found in this search")

        "Your Account",
        "Your Orders",
        "Your Wish List"
    ]

    for expected in expected_items:

        assert any(
            expected.lower() in item.lower()
            for item in menu_texts
        ), f"Missing menu item: {expected}"

# TC07: Count available languages
def test_count_languages(driver):

    driver.get(URL)

    wait = WebDriverWait(driver, 15)

    language_menu = wait.until(
        EC.presence_of_element_located(
            (By.ID, "icp-nav-flyout")
        )
    )

    ActionChains(driver).move_to_element(
        language_menu
    ).perform()

    dropdown = wait.until(
        EC.visibility_of_element_located(
            (By.ID, "nav-flyout-icp")
        )
    )

    # Find language radio buttons
    languages = dropdown.find_elements(
        By.CSS_SELECTOR, "input[name='lop']"
    )

    language_names = []

    for language in languages:

        labels = language.find_elements(
            By.XPATH, "./following-sibling::label"
        )

        if labels:
            name = labels[0].text.strip()

            if name:
                language_names.append(name)

    print("Available languages:")

    for language in language_names:
        print(language)

    print("Total languages:", len(language_names))

    assert len(language_names) > 0, (
        "No languages found in dropdown"
    )
